# The search result contains the correct song as the first entry (title and both artists match).
# Let's automate this for all archived songs: for each, search and pick the top result where the title matches exactly (case-insensitive), and all artists are present.
# We'll build a mapping from (title, artists) to song_id for all archived songs.

archived_song_ids = {}

for song in archived_songs:
    query = f'{song["title"]} {" ".join(song["artists"])}'
    results = apis.spotify.search_songs(access_token=spotify_access_token, query=query, page_limit=5)
    # Find the best match: title matches exactly (case-insensitive), and all artists are present
    match = None
    for result in results:
        if result["title"].strip().lower() == song["title"].strip().lower():
            result_artist_names = set(artist["name"].strip().lower() for artist in result["artists"])
            song_artist_names = set(a.strip().lower() for a in song["artists"])
            if song_artist_names.issubset(result_artist_names):
                match = result
                break
    if match:
        archived_song_ids[(song["title"], tuple(song["artists"]))] = match["song_id"]
    else:
        archived_song_ids[(song["title"], tuple(song["artists"]))] = None  # Not found

print(archived_song_ids)

```python
# Let's fetch all recommended songs for the user. We'll get all pages by looping until we get less than the page_limit (20) results in a page.
recommended_songs = []
page_index = 0
page_limit = 20

while True:
    recs = apis.spotify.show_recommendations(access_token=access_token, page_index=page_index, page_limit=page_limit)
    if not recs or isinstance(recs, dict) and "message" in recs:
        break
    recommended_songs.extend(recs)
    if len(recs) < page_limit:
        break
    page_index += 1

print(f"Total recommended songs: {len(recommended_songs)}")
print(recommended_songs[:3])  # Show a sample of the recommendations for verification.
```

# There are some duplicate songs in the shuffled list (e.g., "Golden Lies", "The Tragedy of Living Without", "City of Fallen Angels").
# Let's remove duplicates while preserving the shuffle order, then add each unique song to the queue one by one.

unique_song_ids = set()
unique_songs = []
for song in all_recommended_songs:
    if song["song_id"] not in unique_song_ids:
        unique_song_ids.add(song["song_id"])
        unique_songs.append(song)

# Now add each unique song to the queue
add_results = []
for song in unique_songs:
    result = apis.spotify.add_to_queue(access_token=access_token, song_id=song["song_id"])
    add_results.append({"song_id": song["song_id"], "result": result})

print(add_results)  # To verify all songs were added successfully.

# The current song has song_id 322, which is not in the set of downloaded_song_ids.
# Let's keep going to the next song until we reach a downloaded one.
# We'll loop, calling next_song and show_current_song, until we find a downloaded song.

access_token = "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJzdWIiOiJzcG90aWZ5K2NhcmJyb3duQGdtYWlsLmNvbSIsImV4cCI6MTY4NDQxMjA5OH0.kf2xG5M8F_mduPWJbWrURnNy6or8nSA7bzIo2UbELCk"
max_steps = 50  # safety limit to avoid infinite loop
steps = 0
song_id = 322

while song_id not in downloaded_song_ids and steps < max_steps:
    next_response = apis.spotify.next_song(access_token=access_token)
    if not next_response or "song_id" not in next_response:
        break
    song_id = next_response["song_id"]
    current_song = apis.spotify.show_current_song(access_token=access_token)
    if not current_song or "song_id" not in current_song:
        break
    song_id = current_song["song_id"]
    steps += 1

# Let's print the final song_id and steps taken.
print({"final_song_id": song_id, "steps_taken": steps})

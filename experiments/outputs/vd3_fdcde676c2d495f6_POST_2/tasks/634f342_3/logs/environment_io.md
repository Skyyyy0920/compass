def get_playlist_songs(playlist_id):
    songs = []
    page_index = 0
    page_limit = 20
    while True:
        songs_page = apis.spotify.show_playlist_songs(access_token=spotify_access_token, playlist_id=playlist_id, page_index=page_index, page_limit=page_limit)
        if not songs_page:
            break
        songs.extend(songs_page)
        if len(songs_page) < page_limit:
            break
        page_index += 1
    return songs

for playlist in all_playlists:
    playlist_id = playlist["playlist_id"]
    playlist_songs = get_playlist_songs(playlist_id)
    for song in playlist_songs:
        # Construct song title by artist(s) string to match old songs list format
        song_title = song.get("title", "").strip()
        artists = song.get("artists", [])
        artist_names = ", ".join(artists)
        song_title_artist = f"{song_title} by {artist_names}"

            songs_to_archive.append({
                "playlist_id": playlist_id,
                "song_id": song["song_id"],
                "song_title_artist": song_title_artist
            })

print(f"Total old songs found in playlists: {len(songs_to_archive)}")

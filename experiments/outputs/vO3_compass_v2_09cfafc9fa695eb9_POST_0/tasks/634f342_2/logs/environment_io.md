# Parse the archive file content to extract song titles and artists
import re

content = archive_file['content']
# Extract lines that start with "- "
song_lines = [line.strip()[2:] for line in content.split('\n') if line.strip().startswith('- ')]
# Each line is "Title by Artist"
archived_songs = []
for line in song_lines:
    # Split on " by " (last occurrence)
    parts = line.rsplit(' by ', 1)
    if len(parts) == 2:
        title = parts[0].strip()
        artist = parts[1].strip()
        archived_songs.append((title, artist))
print(archived_songs)

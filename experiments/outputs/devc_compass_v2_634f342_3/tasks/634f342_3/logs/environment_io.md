# I have the Spotify access token. Now I will get all the user's playlists (both public and private) by paging through the results.
spotify_access_token = spotify_login_response["access_token"]

all_playlists = []
page_index = 0
page_limit = 20

while True:
    playlists_page = apis.spotify.show_playlist_library(access_token=spotify_access_token, page_index=page_index, page_limit=page_limit)
    if not playlists_page:
        break
    all_playlists.extend(playlists_page)
    if len(playlists_page) < page_limit:
        break
    page_index += 1

print(f"Total playlists found: {len(all_playlists)}")  # To verify the number of playlists retrieved.

    playlist_details = apis.spotify.show_playlist(playlist_id=playlist_id, access_token=spotify_access_token)
    if "songs" not in playlist_details:

    songs_in_playlist = playlist_details["songs"]
    to_remove = []
    for song in songs_in_playlist:
        if song["title"] in old_songs:
            to_remove.append(song["id"])
            songs_to_archive.append(song["id"])
    if to_remove:
        songs_to_remove_by_playlist[playlist_id] = to_remove

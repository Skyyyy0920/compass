# Great! Now let's automate this for all the songs with RELEASED_AT TODO.
# We'll search each song on Spotify, extract the release month/year, and build a mapping of song title to release month/year.
release_info = {}
for song in todo_songs:
    query = song["title"] + " " + " ".join(song["artists"])
    results = apis.spotify.search_songs(query=query, page_limit=3)
    if results and results[0]["title"].lower() == song["title"].lower():
        release_date = results[0]["release_date"]
        dt = datetime.strptime(release_date, "%Y-%m-%dT%H:%M:%S")
        release_info[song["title"]] = dt.strftime("%m/%Y")
    else:
        release_info[song["title"]] = None  # Could not find
print(release_info)

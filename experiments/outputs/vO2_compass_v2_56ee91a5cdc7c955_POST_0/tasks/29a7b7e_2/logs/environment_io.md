```python
# Let's parse the file entries to understand the structure.
import re

# Filter out directories (entries ending with '/')
file_entries = [e for e in entries if not e.endswith('/')]
print("File entries:")
for f in file_entries:
    print(f)

# Parse each file name
parsed = []
for f in file_entries:
    # Extract filename from path
    filename = f.split('/')[-1]
    # Pattern: <date>__<file_name>.<extension>
    match = re.match(r'(.+?)__(.+)\.(.+)$', filename)
    if match:
        date = match.group(1)
        file_name = match.group(2)
        extension = match.group(3)
        parsed.append({'date': date, 'file_name': file_name, 'extension': extension, 'source': f})
        print(f"date={date}, file_name={file_name}, ext={extension}")
    else:
        print(f"Could not parse: {filename}")
```

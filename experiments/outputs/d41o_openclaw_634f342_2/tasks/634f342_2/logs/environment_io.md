# Let's extract the song titles and artists from the file content. We'll ignore the first line (header) and parse the rest.
import re

lines = songs_file["content"].splitlines()
songs_to_archive = []
for line in lines:
    match = re.match(r"- (.+) by (.+)", line.strip())
    if match:
        title, artist = match.groups()
        songs_to_archive.append({"title": title.strip(), "artist": artist.strip()})
print(songs_to_archive)

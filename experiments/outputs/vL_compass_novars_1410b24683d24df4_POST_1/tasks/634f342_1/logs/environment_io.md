```python
# Let's parse the content of the file to extract the song titles and artists.
import re

old_songs_content = old_songs_file["content"]
# Each song is on a line like: "- Song Title by Artist"
song_lines = [line.strip() for line in old_songs_content.splitlines() if line.strip().startswith("- ")]
# Extract song title and artist using regex
old_songs = []
for line in song_lines:
    match = re.match(r"- (.+) by (.+)", line)
    if match:
        title, artist = match.groups()
        old_songs.append({"title": title.strip(), "artist": artist.strip()})

print(old_songs)
```

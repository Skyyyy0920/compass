```python
# Get all playlists by paginating through all pages.
all_playlists = []
page_index = 0
page_limit = 20
while True:
    page = apis.spotify.show_playlist_library(access_token=access_token, page_index=page_index, page_limit=page_limit)
    if not page:
        break
    all_playlists.extend(page)
    if len(page) < page_limit:
        break
    page_index += 1

print(f"Total playlists: {len(all_playlists)}")
for p in all_playlists:
    print(p['playlist_id'], p['title'], p['song_ids'])
```

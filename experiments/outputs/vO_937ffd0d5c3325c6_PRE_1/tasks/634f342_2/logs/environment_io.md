songs_to_archive = []
for line in song_lines:
    # Format: "- Title by Artist"
    # Remove leading "- "
    line = line[2:].strip()
    # Split on " by " (last occurrence)
    if ' by ' in line:
        title, artist = line.rsplit(' by ', 1)
        songs_to_archive.append({'title': title.strip(), 'artist': artist.strip()})
    else:
        # If no " by ", treat whole as title
        songs_to_archive.append({'title': line, 'artist': ''})

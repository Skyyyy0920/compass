```python
# Now I have the access token. Let's fetch all recommended songs for Glen Moore.
# The API paginates results, so I will fetch all pages until no more songs are returned.

recommended_songs = []
page_index = 0
page_limit = 20

while True:
    recs = apis.spotify.show_recommendations(
        access_token=access_token,
        page_index=page_index,
        page_limit=page_limit
    )
    if not recs or len(recs) == 0:
        break
    recommended_songs.extend(recs)
    if len(recs) < page_limit:
        break
    page_index += 1

print(recommended_songs)
```

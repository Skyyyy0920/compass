```python
# I need to pass the access_token parameter to the show_liked_songs API.
# Also, I need to handle pagination to get all liked songs.
all_liked_songs = []
page_index = 0
page_limit = 20  # max allowed

while True:
    liked_songs_page = apis.spotify.show_liked_songs(access_token=spotify_access_token, page_index=page_index, page_limit=page_limit)
    if not liked_songs_page:
        break
    all_liked_songs.extend(liked_songs_page)
    if len(liked_songs_page) < page_limit:
        break
    page_index += 1
```

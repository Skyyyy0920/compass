### Environment Interaction 8
----------------------------------------------------------------------------
```python
# I will search for top songs in each genre sorted by play_count descending to simulate recommended songs.
# I will collect up to 5 songs per genre to keep the queue manageable.
recommended_songs = []
for genre in genres:
    response = apis.spotify.search_songs(genre=genre, sort_by='-play_count', page_index=0, page_limit=5)
    if response:
        recommended_songs.extend(response)

print(f"Collected {len(recommended_songs)} recommended songs.")
```

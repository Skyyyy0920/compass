# Parse the old songs file to get list of (title, artists) tuples
import re

old_songs_content = old_songs_file['content']
old_songs = []
for line in old_songs_content.split('\n'):
    line = line.strip()
    if line.startswith('- '):
        # Remove the "- " prefix
        song_str = line[2:]
        # Split on " by " to get title and artists
        if ' by ' in song_str:
            title, artists_str = song_str.split(' by ', 1)
            artists = [a.strip() for a in artists_str.split(',')]
            old_songs.append({'title': title.strip(), 'artists': artists})

print(f"Parsed {len(old_songs)} old songs")

    print(s)

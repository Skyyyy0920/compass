```python
# The show_liked_songs API requires the access_token parameter.
# I will call show_liked_songs with the access_token obtained from login.
access_token = spotify_login_response["access_token"]

# I will fetch all liked songs by paginating through all pages.
liked_songs_all = []
page_index = 0
while True:
    liked_songs_page = apis.spotify.show_liked_songs(access_token=access_token, page_index=page_index, page_limit=20)
    if not liked_songs_page:
        break
    liked_songs_all.extend(liked_songs_page)
    if len(liked_songs_page) < 20:
        break
    page_index += 1

print(f"Total liked songs fetched: {len(liked_songs_all)}")
```

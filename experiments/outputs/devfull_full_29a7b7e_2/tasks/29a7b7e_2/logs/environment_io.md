# I have the list of files in the directory.
# Now I will parse each file name from the format "<date>__<file_name>.<extension>" and reorganize them into "<file_name>/<date>.<extension>".
# For each file, I will:
# 1. Extract the date, file_name, and extension.
# 2. Create the directory "<file_name>" if it does not exist.
# 3. Move the file to "<file_name>/<date>.<extension>".
import os

files = directory_files
access_token = login_response["access_token"]

for file_path in files:
    # Extract the base file name
    base_name = os.path.basename(file_path)
    # Split into date and rest by "__"
    date_part, rest = base_name.split("__", 1)
    # Split rest into file_name and extension
    file_name, extension = os.path.splitext(rest)
    extension = extension.lstrip(".")
    # New directory path
    new_dir = f"~/documents/work/meetings_files/{file_name}"
    # New file path
    new_file_path = f"{new_dir}/{date_part}.{extension}"
    # Create directory if not exists
    apis.file_system.create_directory(access_token=access_token, directory_path=new_dir, recursive=True)
    # Move the file to new location
    apis.file_system.move_file(access_token=access_token, source_path=file_path, destination_path=new_file_path)

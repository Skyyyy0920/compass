```python
# The note contains a list of songs, some of which already have RELEASED_AT dates, while others have "RELEASED_AT TODO".
# Let's extract the songs with "RELEASED_AT TODO" so we can look up their release months.
import re

note_content = note["content"]
todo_songs = []
for line in note_content.splitlines():
    match = re.match(r"- (.+) BY (.+) RELEASED_AT TODO", line)
    if match:
        song_title = match.group(1).strip()
        artist = match.group(2).strip()
        todo_songs.append({"title": song_title, "artist": artist})

print(todo_songs)
```

# The note uses lines like:
# - Song Title BY Artist1, Artist2 RELEASED_AT MM/YYYY
# or
# - Song Title BY Artist1, Artist2 RELEASED_AT TODO
# Let's extract all lines with RELEASED_AT TODO, and for each, parse the song title and artist(s).
import re

note_content = note["content"]
todo_songs = []

for line in note_content.splitlines():
    match = re.match(r"- (.+) BY (.+) RELEASED_AT TODO", line.strip())
    if match:
        song_title = match.group(1).strip()
        artist = match.group(2).strip()
        todo_songs.append({"title": song_title, "artist": artist, "line": line})

print(todo_songs)

# Now, for each song in todo_songs, let's search Spotify using the song title and artist to get the release_date.
# We'll use the query: "<song title> <artist>" for best match.
# We'll collect the release month (MM/YYYY) for each song.
release_months = {}

for song in todo_songs:
    query = f"{song['title']} {song['artist']}"
    results = apis.spotify.search_songs(query=query, page_limit=1)
    if results and isinstance(results, list) and len(results) > 0:
        release_date = results[0].get("release_date")
        if release_date:
            # release_date is in "YYYY-MM-DDTHH:MM:SS"
            year, month = release_date[:4], release_date[5:7]
            release_months[song['line']] = f"{month}/{year}"
        else:
            release_months[song['line']] = None
    else:
        release_months[song['line']] = None

print(release_months)

note_content = """# Spotify Liked Songs

- Midnight Reverie BY Aria Sterling
"""

import re

# Parse each line for song and artist, and check for RELEASED_AT
song_entries = []
for line in note_content.splitlines():
    m = re.match(r'- (.+?) BY (.+?)(?: RELEASED_AT ([0-9]{2}/[0-9]{4}))?$', line.strip())
    if m:
        title = m.group(1).strip()
        artists = [a.strip() for a in m.group(2).split(',')]
        released_at = m.group(3)
        song_entries.append({
            "title": title,
            "artists": artists,
            "released_at": released_at
        })

# Find songs missing release month info

### Environment Interaction 9
----------------------------------------------------------------------------
```python
# I need to pass the access_token parameter to the show_liked_songs API.
# Also, I need to handle pagination to get all liked songs.
liked_songs_all = []
page_index = 0
page_limit = 20  # max allowed per page

while True:
    response = apis.spotify.show_liked_songs(access_token=spotify_access_token, page_index=page_index, page_limit=page_limit)
    if not response:
        break
    liked_songs_all.extend(response)
    if len(response) < page_limit:
        break
    page_index += 1

liked_songs_all  # Check the full list of liked songs.
```

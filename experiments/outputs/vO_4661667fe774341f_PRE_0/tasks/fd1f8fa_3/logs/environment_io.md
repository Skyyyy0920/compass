```python
access_token = login_result['access_token']

# Get liked songs (all pages)
liked_songs = []
page_index = 0
while True:
    page = apis.spotify.show_liked_songs(access_token=access_token, page_index=page_index, page_limit=20)
    if not page:
        break
    liked_songs.extend(page)
    if len(page) < 20:
        break
    page_index += 1

print("Liked songs count:", len(liked_songs))
liked_song_ids = {song['song_id'] for song in liked_songs}
print("Liked song IDs:", liked_song_ids)

# Get song queue
queue = apis.spotify.show_song_queue(access_token=access_token)
print("Queue length:", len(queue))
for song in queue:
    print(song['position'], song['song_id'], song['title'])
```

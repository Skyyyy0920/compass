# The "show_liked_songs" API returns the user's liked songs, which can be used as a proxy for recommended songs.
# Let's fetch all liked songs (using pagination) to use them for resetting the queue.
liked_songs = []
page_index = 0
page_limit = 20

while True:
    result = apis.spotify.show_liked_songs(access_token=access_token, page_index=page_index, page_limit=page_limit)
    if not result or isinstance(result, dict) and "message" in result:
        break
    liked_songs.extend(result)
    if len(result) < page_limit:
        break
    page_index += 1

print(liked_songs)

# Let's parse the file content to extract the song titles and artists.
file_content = old_songs_file["content"]

# Remove the header and split into lines, then extract song info from each line.
import re

old_songs = []
for line in file_content.splitlines():
    match = re.match(r"- (.+?) by (.+)", line.strip())
    if match:
        title = match.group(1).strip()
        artists = [artist.strip() for artist in match.group(2).split(",")]
        old_songs.append({"title": title, "artists": artists})

print(old_songs)

```python
# The show_liked_songs API requires the access_token parameter.
# I will call show_liked_songs with the access_token obtained from login.
# Also, I will handle pagination to get all liked songs.

access_token = login_response["access_token"]

all_liked_songs = []
page_index = 0
page_limit = 20

while True:
    liked_songs_page = apis.spotify.show_liked_songs(access_token=access_token, page_index=page_index, page_limit=page_limit)
    if not liked_songs_page:
        break
    all_liked_songs.extend(liked_songs_page)
    if len(liked_songs_page) < page_limit:
        break
    page_index += 1

print(f"Total liked songs fetched: {len(all_liked_songs)}")
```

# Update the note content with release months
note_content = note['content']
lines = note_content.split('\n')
new_lines = []
for line in lines:
    if 'RELEASED_AT TODO' in line:
        # Extract song title: between "- " and " BY "
        # Example: "- Lost in the Embrace of Night BY Noah Bennett RELEASED_AT TODO"
        # Find the part after "- " and before " BY "
        start = line.find('- ') + 2
        end = line.find(' BY ', start)
        title = line[start:end].strip()
        if title in song_title_to_release:
            release_month = song_title_to_release[title]
            line = line.replace('RELEASED_AT TODO', f'RELEASED_AT {release_month}')
        else:
            print(f"Warning: No release month found for title: {title}")
    new_lines.append(line)
new_content = '\n'.join(new_lines)
print(new_content)

# Parse the songs to archive from the file content
import re
songs_to_archive = []
for line in file_content['content'].split('\n'):
    line = line.strip()
    if line.startswith('- '):
        song_info = line[2:]
        # Format: "Song Title by Artist"
        match = re.match(r'(.+) by (.+)', song_info)
        if match:
            songs_to_archive.append({'title': match.group(1).strip(), 'artist': match.group(2).strip()})
print(songs_to_archive)

# Get all playlists from the user's library (handle pagination)
all_playlists = []

while True:
    playlists_page = apis.spotify.show_playlist_library(access_token=spotify_login['access_token'], page_index=page_index, page_limit=20)
    if not playlists_page:
        break
    all_playlists.extend(playlists_page)
    if len(playlists_page) < 20:
        break
    page_index += 1
print(f"Total playlists: {len(all_playlists)}")
for p in all_playlists:

# Let's search for each archive song to get its Spotify ID
# We already have the first song's ID (117), but let's search for all to be safe
archive_song_ids = []

for song_info in archive_songs:
    title, artist = song_info
    query = f"{title} {artist}"
    search_result = apis.spotify.search_songs(query=query, access_token=spotify_login['access_token'], page_limit=20)
    # Find the matching song
    found = False
    if isinstance(search_result, list):
        for song in search_result:
            if song['title'].lower() == title.lower():
                # Check if artist matches
                for art in song['artists']:
                    if art['name'].lower() == artist.lower():
                        archive_song_ids.append(song['song_id'])
                        found = True
                        break
                if found:
                    break
    if not found:
        print(f"Could not find song: {title} by {artist}")
    else:
        print(f"Found: {title} by {artist} -> ID {archive_song_ids[-1]}")

print(f"Total archive song IDs found: {len(archive_song_ids)}")

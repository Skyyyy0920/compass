# Let's parse the note content to extract songs with TODO release months
import re

content = note['content']
lines = content.strip().split('\n')
# Skip the header line
song_lines = [line for line in lines if line.startswith('- ')]

# Parse each song line
songs = []
for line in song_lines:
    # Format: - Song Title BY Artist1, Artist2 RELEASED_AT MM/YYYY or TODO
    match = re.match(r'- (.+) BY (.+) RELEASED_AT (.+)', line)
    if match:
        title = match.group(1).strip()
        artists_str = match.group(2).strip()
        release = match.group(3).strip()
        artists = [a.strip() for a in artists_str.split(',')]
        songs.append({'title': title, 'artists': artists, 'release': release, 'line': line})

# Show songs with TODO
todo_songs = [s for s in songs if s['release'] == 'TODO']
print(f"Total songs: {len(songs)}, TODO songs: {len(todo_songs)}")
for s in todo_songs:
    print(f"  {s['title']} BY {', '.join(s['artists'])}")

lines = content.split('\n')
new_lines = []
for line in lines:
    if 'RELEASED_AT TODO' in line:
        # Extract song title from the line: format "- <title> BY <artists> RELEASED_AT TODO"
        # The title is between "- " and " BY "
        title = line.split(' BY ')[0].lstrip('- ').strip()
        if title in release_map:
            line = line.replace('RELEASED_AT TODO', f'RELEASED_AT {release_map[title]}')

            print(f"Warning: no release month for {title}")
    new_lines.append(line)

new_content = '\n'.join(new_lines)

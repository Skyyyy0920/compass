```python
# Let's parse the content of the file to extract the song titles and artists.
import re

old_songs_content = old_songs_file["content"]
# Each song is on a line like: "- Song Title by Artist"
song_lines = [line.strip() for line in old_songs_content.splitlines() if line.strip().startswith("- ")]
# Extract song title and artist using regex
old_songs = []
for line in song_lines:
    match = re.match(r"- (.+) by (.+)", line)
    if match:
        title, artist = match.groups()
        old_songs.append({"title": title.strip(), "artist": artist.strip()})

print(old_songs)
```

```python
# The top result matches the title and both artists exactly, so song_id 74 is the correct Spotify ID for "On the Border of Reality".
# Let's automate this search for all old songs, collecting the best match (exact title and all artists) for each.

def find_best_match(song, search_results):
    # Match title exactly (case-insensitive, strip), and all artists (ignoring order and whitespace)
    song_title = song["title"].strip().lower()
    song_artists = set(a.strip().lower() for a in song["artist"].split(","))
    for result in search_results:
        result_title = result["title"].strip().lower()
        result_artists = set(a["name"].strip().lower() for a in result["artists"])
        if result_title == song_title and song_artists == result_artists:
            return result["song_id"]
    # If no exact match, return None
    return None
```

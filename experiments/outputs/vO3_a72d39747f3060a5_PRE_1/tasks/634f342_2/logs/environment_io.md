while True:
    playlists_page = apis.spotify.show_playlist_library(access_token=spotify_token, page_index=page_index, page_limit=20)
    if not playlists_page:
        break
    all_playlists.extend(playlists_page)
    if len(playlists_page) < 20:
        break
    page_index += 1

# Parse the songs to archive from the file content
file_content = file_data['content']
lines = file_content.split('\n')
songs_to_archive = []
for line in lines:
    line = line.strip()
    if line.startswith('- '):
        song_str = line[2:].strip()
        # Split by " by " from the right to get title and artist
        if ' by ' in song_str:
            title, artist = song_str.rsplit(' by ', 1)
            songs_to_archive.append((title.strip(), artist.strip()))
        else:
            # If no artist, just title
            songs_to_archive.append((song_str.strip(), None))

print(f"Parsed {len(songs_to_archive)} songs to archive:")
for t, a in songs_to_archive:
    print(f"  Title: {t}, Artist: {a}")

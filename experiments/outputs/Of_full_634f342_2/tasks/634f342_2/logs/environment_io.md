# Let's find the song IDs for all songs in the archive list by searching for each song.
# We'll search by title and match the artist name.

archive_song_ids = []

for song in songs_to_archive:
    title = song['title']
    artist = song['artist']
    
    # Search for the song by title
    results = apis.spotify.search_songs(query=title, access_token=spotify_token, page_limit=20)
    
    # Find the matching song with the correct artist
    found = False
    for result in results:
        if result['title'] == title:
            # Check if any artist matches
            for a in result['artists']:
                if a['name'] == artist:
                    archive_song_ids.append(result['song_id'])
                    found = True
                    break
        if found:
            break
    
    if not found:
        print(f"Could not find song: {title} by {artist}")

print("Archive song IDs:", archive_song_ids)
print("Count:", len(archive_song_ids))

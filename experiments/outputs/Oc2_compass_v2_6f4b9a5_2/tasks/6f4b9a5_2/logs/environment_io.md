    "Final Echo": "07/2022"
}

lines = content.split('\n')
new_lines = []
for line in lines:
    if 'RELEASED_AT TODO' in line:
        # Extract song title: part before " BY "
        song_part = line.split(' BY ')[0].replace('- ', '').strip()
        # Find the release month
        month = release_months.get(song_part)
        if month:
            line = line.replace('RELEASED_AT TODO', f'RELEASED_AT {month}')
        else:
            print(f"Warning: no release month found for {song_part}")
    new_lines.append(line)

new_content = '\n'.join(new_lines)
print(new_content)

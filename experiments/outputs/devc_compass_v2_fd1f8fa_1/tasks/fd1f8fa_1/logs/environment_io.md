### Environment Interaction 7
----------------------------------------------------------------------------
```python
# I will now fetch all liked songs by paging through the results (max 20 per page).
liked_songs = []
page_index = 0
while True:
    page = apis.spotify.show_liked_songs(access_token=spotify_login_response["access_token"], page_index=page_index, page_limit=20)
    if not page:
        break
    liked_songs.extend(page)
    if len(page) < 20:
        break
    page_index += 1
print(f"Total liked songs fetched: {len(liked_songs)}")
```

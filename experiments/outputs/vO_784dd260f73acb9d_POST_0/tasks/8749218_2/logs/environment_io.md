```python
access_token = login_response["access_token"]
# Get all recommendations
all_recommendations = []
page_index = 0
page_limit = 20
while True:
    page = apis.spotify.show_recommendations(access_token=access_token, page_index=page_index, page_limit=page_limit)
    if not page:
        break
    all_recommendations.extend(page)
    if len(page) < page_limit:
        break
    page_index += 1
print(f"Total recommendations: {len(all_recommendations)}")
print(all_recommendations[:3])
```

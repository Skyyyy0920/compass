### Environment Interaction 15
----------------------------------------------------------------------------
```python
phone_access_token = phone_token["access_token"]
venmo_access_token = venmo_token["access_token"]

def get_all_contacts(token):
    contacts = []
    page_index = 0
    page_limit = 20
    while True:
        page = apis.phone.search_contacts(access_token=token, page_index=page_index, page_limit=page_limit)
        if not page:
            break
        contacts.extend(page)
        if len(page) < page_limit:
            break
        page_index += 1
    return contacts

def get_all_venmo_friends(token):
    friends = []
    page_index = 0
    page_limit = 20
    while True:
        page = apis.venmo.search_friends(access_token=token, page_index=page_index, page_limit=page_limit)
        if not page:
            break
        friends.extend(page)
        if len(page) < page_limit:
            break
        page_index += 1
    return friends

contacts = get_all_contacts(phone_access_token)
```

```python
# I have the Spotify access_token. Now I will get all playlists from my Spotify library, considering pagination.
spotify_access_token = spotify_login_response["access_token"]

all_playlists = []
page_index = 0
page_limit = 20

while True:
    response = apis.spotify.show_playlist_library(access_token=spotify_access_token, page_index=page_index, page_limit=page_limit)
    if isinstance(response, dict) and "message" in response:
        # Failure case
        break
    all_playlists.extend(response)
    if len(response) < page_limit:
        # No more pages
        break
    page_index += 1

print(f"Total playlists found: {len(all_playlists)}")
```
